                                 const currentDate = new Date(day.date + 'T00:00:00'); // Ensure parsing as local date

                                     if (minDate === null || currentDate < minDate) {

                                         minDate = currentDate;

                         }

                     });

                 });

             };

                                     const date = new Date(day.date + 'T00:00:00');

                                     if (!isNaN(date)) return date;

                                 } catch (e) { console.warn(`Skipping invalid date: ${day.date}`);}

                             }

                         }

                     }

                 }

                 return new Date(); // Fallback to today if no valid date found

             };

                     } else {

                         // Next month's days

                     'END:VCALENDAR'

                 ].join('\r\n');
